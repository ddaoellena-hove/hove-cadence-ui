{/* Toutes les valeurs affichées sont lues dans les variables CSS de tokens.css
    (getComputedStyle) : aucune valeur n'est recopiée ici. */}

export const useTokenValue = (name) => {
  const ref = useRef(null);
  const [value, setValue] = useState("");
  useEffect(() => {
    const el = ref.current ?? document.documentElement;
    setValue(getComputedStyle(el).getPropertyValue(name).trim());
  }, [name]);
  return [ref, value];
};

export const TokenValue = ({ name }) => {
  const [ref, value] = useTokenValue(name);
  return (
    <span ref={ref} className="tk-value">
      {value || "—"}
    </span>
  );
};

      </div>
    </td>
    <td>
      <div className="tk-pair">

      </div>
